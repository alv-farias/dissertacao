\begin{simbolos} \itemsep -1pt
	\item[$V(G)$] Conjunto de vértices do grafo $G$
	\item[$E(G)$] Conjunto de arestas do grafo $G$
	\item[$N_G(v)$] Vizinhança do vértice $v$ no grafo $G$
	\item[{\VizFec}] Vizinhança fechada do vértice $v$ no grafo $G$
	\item[$N^+_D(v)$] Vizinhança de saída do vértice $v$ no grafo direcionado $D$
	\item[$N^-_D(v)$] Vizinhança de entrada do vértice $v$ no grafo direcionado $D$
	\item[$d_G(v)$] Grau do vértice $v$ no grafo $G$
	\item[$d^+_D(v)$] Grau de saída do vértice $v$ no grafo direcionado $D$
	\item[$d^-_D(v)$] Grau de entrada do vértice $v$ no grafo direcionado $D$
    \item[{\SubInd}] Subgrafo de $G$ induzido por $S$
    \item[$d_G(u,v)$] Distância entre os vértices $u$ e $v$ no grafo $G$
	\item[$T(G)$] Árvore de blocos e vértices de corte do grafo $G$
    \item[$conv_\mathcal{C}(S)$] Envoltória convexa de $S$ na convexidade $\mathcal{C}$
    \item[$Ext_\mathcal{C}(S)$] Conjunto dos vértices extremos do convexo $S$ em $\mathcal{C}$
    \item[$hn_\mathcal{C}(G)$] Número de envoltória do grafo $G$ na convexidade $\mathcal{C}$
    \item[$in_\mathcal{C}(G)$] Número de intervalo do grafo $G$ na convexidade $\mathcal{C}$
    \item[$con_\mathcal{C}(G)$] Número de convexidade do grafo $G$ na convexidade $\mathcal{C}$
    \item[$cth_\mathcal{C}(G)$] Número de Carathéodory do grafo $G$ na convexidade $\mathcal{C}$
    \item[$h\ell _\mathcal{C}(G)$] Número de Helly do grafo $G$ na convexidade $\mathcal{C}$
    \item[$rd_\mathcal{C}(G)$] Número de Radon do grafo $G$ na convexidade $\mathcal{C}$
    \item[$rk_\mathcal{C}(G)$] Posto (Rank) do grafo $G$ na convexidade $\mathcal{C}$
    \item[$gp_\mathcal{C}(G)$] Número de posição geral do grafo $G$ na convexidade $\mathcal{C}$
    \item[$b(G)$] Quantidade de vértices do menor bloco-folha do grafo $G$
\end{simbolos}
